// Function called by BOTH the Load button AND the QR Scanner async function loadEquipmentDetails(pidToLoad) { // 1. Get PID from argument (QR Scanner) or from input box (Manual Entry) const inputEl = document.getElementById('pidInput'); const pid = (pidToLoad || inputEl.value).trim(); if (!pid) return; // Show loading indicator document.getElementById('machineCard').classList.add('opacity-50'); try { // 2. Fetch live machine data from Apps Script const response = await fetch(`${API_URL}?action=getMachineDetails&pid=${encodeURIComponent(pid)}`); const data = await response.json(); if (data.error) { alert('Machine not found: ' + pid); return; } // 3. Update the UI dynamically document.getElementById('displayPid').textContent = data.pid.toUpperCase(); document.getElementById('displayTitle').textContent = data.info['Equipment Name'] || data.info['Type'] || 'Equipment'; document.getElementById('displayLocation').textContent = data.info['Location'] || 'Facility'; // 4. Render Maintenance Checklist renderChecklist(data.maint); // 5. Render Timeline History renderTimeline(data.history); } catch (err) { console.error('Failed to load machine data:', err); } finally { document.getElementById('machineCard').classList.remove('opacity-50'); } } // Bind Enter Key press inside the input box document.getElementById('pidInput').addEventListener('keypress', function (e) { if (e.key === 'Enter') { loadEquipmentDetails(); } });
